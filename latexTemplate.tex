% puts the text to the right of the line if enough space
% else on right side of a new line
%%	#1: text to put to the right (here used for points)
\newcommand\PointsSpacing[1]{{%
	\unskip\nobreak\hfil\penalty50
	\hskip2em\hbox{}\nobreak\hfil #1%
	\parfillskip=0pt \finalhyphendemerits=0 \par}}

% PyRopeExercise command
%%	#1: preamble
%%	#2: problem
%%	#3: max total score
\newcommand{\PyRopeExercise}[3]{\item
	\expandafter\ifx\expandafter\relax
	\detokenize{#1}\relax\else #1 \\ \fi #2 %
	\PointsSpacing{\rule{10pt}{0.4pt}~/ #3 P.}}

% PyRopeHint command
%%	#1: hint text
\newcommand{\PyRopeHint}[1]{\noindent\fbox{%
	\parbox{\dimexpr%
	\linewidth-2\fboxsep-2\fboxrule}%
	{#1}}\vspace{1mm}}

% PyRopeDropdown command
%%	overall dropdown style definition
%%	#1: dropdown options list; options separated by semicolon
\NewDocumentCommand\PyRopeDropdown{>{\SplitList{;}}m}
{%
	[
	\ProcessList{#1}{\PyRopeDropdownItem}%
	\firstitemtrue
	]
}
\newif\iffirstitem
\firstitemtrue 
% PyRopeDropdownItem command
%%	style definition of one item
%%	#1: dropdown item
\newcommand\PyRopeDropdownItem[1]{%
	\iffirstitem
		\firstitemfalse
	\else
		$\vert$~%
	\fi
	{#1~}}

% PyRopeRadioButtons command
%%	overall list style definition
%%	#1: radiobutton options list; options separated by semicolon
\NewDocumentCommand\PyRopeRadioButtons{>{\SplitList{;}}m}
{%
	\begin{itemize}
		\ProcessList{#1}{\PyRopeRadioButtonsItem}%
	\end{itemize}
}
% PyRopeRadioButtonsItem command
%%	style definition of one item
%%	#1: radio button item
\newcommand{\PyRopeRadioButtonsItem}[1]{%
		\item[$\bigcirc$] #1
}
